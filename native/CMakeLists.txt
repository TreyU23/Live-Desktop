cmake_minimum_required(VERSION 3.24)
project(LiveDesktopWindowsBridge LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with MSVC and a Windows 10/11 SDK containing C++/WinRT headers.")
endif()
add_executable(live-desktop-windows-bridge windows-bridge/main.cpp windows-bridge/capabilities.cpp)
target_compile_features(live-desktop-windows-bridge PRIVATE cxx_std_20)
target_compile_definitions(live-desktop-windows-bridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(live-desktop-windows-bridge PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(live-desktop-windows-bridge PRIVATE windowsapp advapi32 dxgi powrprof shell32 user32)
install(TARGETS live-desktop-windows-bridge RUNTIME DESTINATION bin)
find_path(JNI_HEADERS jni.h HINTS "$ENV{JAVA_HOME}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/jdk-21/include" REQUIRED)
add_library(live-desktop-credentials SHARED credentials/dpapi.cpp)
target_include_directories(live-desktop-credentials PRIVATE "${JNI_HEADERS}" "${JNI_HEADERS}/win32")
target_compile_features(live-desktop-credentials PRIVATE cxx_std_20)
target_compile_definitions(live-desktop-credentials PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(live-desktop-credentials PRIVATE /W4 /EHsc)
target_link_libraries(live-desktop-credentials PRIVATE crypt32)
install(TARGETS live-desktop-credentials RUNTIME DESTINATION bin)

add_executable(live-desktop-launcher launcher/main.cpp)
target_compile_features(live-desktop-launcher PRIVATE cxx_std_20)
target_compile_definitions(live-desktop-launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(live-desktop-launcher PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(live-desktop-launcher PRIVATE advapi32 ws2_32 winhttp iphlpapi)
install(TARGETS live-desktop-launcher RUNTIME DESTINATION bin)

add_executable(live-desktop-host host/main.cpp)
target_compile_features(live-desktop-host PRIVATE cxx_std_20)
target_compile_definitions(live-desktop-host PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(live-desktop-host PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(live-desktop-host PRIVATE ws2_32 winhttp)
install(TARGETS live-desktop-host RUNTIME DESTINATION bin)
enable_testing()
add_test(NAME bridge-protocol COMMAND live-desktop-windows-bridge --self-test)
